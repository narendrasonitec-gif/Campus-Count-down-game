    <!-- ==================== JAVASCRIPT GAME ENGINE ==================== -->
    <script>
        /* ==================== AUDIO SYNTHESIZER ==================== */
        class SoundEngine {
            constructor() {
                this.muted = false;
                this.audioCtx = null;
            }
            init() {
                if (!this.audioCtx) {
                    const AudioContext = window.AudioContext || window.webkitAudioContext;
                    this.audioCtx = new AudioContext();
                }
                if (this.audioCtx.state === 'suspended') {
                    this.audioCtx.resume();
                }
            }
            playTone(freq, duration, type = 'sine', gainVal = 0.1) {
                if (this.muted) return;
                try {
                    this.init();
                    const osc = this.audioCtx.createOscillator();
                    const gain = this.audioCtx.createGain();
                    osc.type = type;
                    osc.frequency.setValueAtTime(freq, this.audioCtx.currentTime);
                    gain.gain.setValueAtTime(gainVal, this.audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, this.audioCtx.currentTime + duration);
                    osc.connect(gain);
                    gain.connect(this.audioCtx.destination);
                    osc.start();
                    osc.stop(this.audioCtx.currentTime + duration);
                } catch(e) {}
            }
            click() { this.playTone(600, 0.08, 'triangle', 0.08); }
            coin() {
                if (this.muted) return;
                try {
                    this.init();
                    const now = this.audioCtx.currentTime;
                    const osc = this.audioCtx.createOscillator();
                    const gain = this.audioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(987.77, now);
                    osc.frequency.setValueAtTime(1318.51, now + 0.08);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.3);
                    osc.connect(gain);
                    gain.connect(this.audioCtx.destination);
                    osc.start(now);
                    osc.stop(now + 0.3);
                } catch(e){}
            }
            success() {
                if (this.muted) return;
                try {
                    this.init();
                    const now = this.audioCtx.currentTime;
                    [523.25, 659.25, 783.99, 1046.50].forEach((f, i) => {
                        const osc = this.audioCtx.createOscillator();
                        const gain = this.audioCtx.createGain();
                        osc.type = 'triangle';

        function updateJoystick(clientX, clientY) {
            const dx = clientX - touchStartX;
            const dy = clientY - touchStartY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            const maxDist = 45;
            const angle = Math.atan2(dy, dx);
            const cappedDist = Math.min(dist, maxDist);
            const thumbX = Math.cos(angle) * cappedDist;
            const thumbY = Math.sin(angle) * cappedDist;

            joystickThumb.style.transform = `translate(${thumbX}px, ${thumbY}px)`;
            joystickDeltaX = thumbX / maxDist;
            joystickDeltaY = thumbY / maxDist;
        }

        /* ==================== TIME & CURFEW LOGIC ==================== */
        let timeAccumulator = 0;
        function updateGameTime(dt) {
            timeAccumulator += dt;
            if (timeAccumulator >= 1.0) {
                timeAccumulator = 0;
                gameState.minute += 2;
                if (gameState.minute >= 60) {
                    gameState.minute = 0;
                    gameState.hour += 1;
                    gameState.energy = Math.max(0, gameState.energy - 1);

                    if (gameState.hour === 20 && gameState.minute === 0) {
                        checkHostelCurfew();
                    }
                    if (gameState.hour >= 23) {
                        advanceToNextDay();
                    }
                }
            }
        }

        function checkHostelCurfew() {
            const hostel = BUILDINGS.find(b => b.id === 'hostel');
            const dist = Math.hypot(gameState.playerX - (hostel.x + hostel.w/2), gameState.playerY - (hostel.y + hostel.h/2));
            if (dist > 300) {
                gameState.wardenWarnings++;
                gameState.wallet = Math.max(0, gameState.wallet - 50);
                gameState.wardenFineTotal += 50;
                gameState.xp = Math.max(0, gameState.xp - 20);
                sounds.alarm();
                showToast(`Warden Curfew Violation! Narendra fined ₹50 & Warning #${gameState.wardenWarnings}`, 'error');
            } else {
                showToast('Safely inside hostel before curfew! Warden pleased.', 'success');
                sounds.success();
            }
        }

        function advanceToNextDay() {
            gameState.day++;
            gameState.hour = 8;
            gameState.minute = 0;
            gameState.dailySpent = 0;

            if (gameState.day === 3) gameState.level = 2;
            if (gameState.day === 5) gameState.level = 3;
            if (gameState.day > 7) {
                triggerVictory();
                return;
            }

            const dailyAllowance = LEVEL_CONFIG[gameState.level].budget;
            gameState.wallet += dailyAllowance;
            gameState.dailyLimit = dailyAllowance;

            showToast(`Day ${gameState.day} started! Daily allowance ₹${dailyAllowance} credited.`, 'success');
            sounds.coin();
            saveGame();
        }

        function triggerVictory() {
            gameState.screen = 'MENU';
            showToast('Congratulations! Narendra successfully completed Campus Countdown!', 'success');
            sounds.success();
        }

        /* ==================== GAME LOOP & RENDERING ==================== */
        let lastTime = performance.now();
        let activeBuilding = null;

        function gameLoop(timestamp) {
            const dt = (timestamp - lastTime) / 1000;
            lastTime = timestamp;

            if (gameState.screen === 'PLAYING') {
                updateGameTime(dt);

                if (gameState.currentScene === 'WORLD') {
                    updatePlayerWorld();
                    updateNPCs();
                    updateVehicles();
                    renderWorld();
                } else {
                    updatePlayerInterior();
                    renderInterior();
                }

                updateHUD();
            }

            requestAnimationFrame(gameLoop);
        }

        function updatePlayerWorld() {
            let dx = 0;
            let dy = 0;

            if (keys['w'] || keys['arrowup']) dy -= 1;
            if (keys['s'] || keys['arrowdown']) dy += 1;
            if (keys['a'] || keys['arrowleft']) dx -= 1;
            if (keys['d'] || keys['arrowright']) dx += 1;

            if (touchActive && (Math.abs(joystickDeltaX) > 0.1 || Math.abs(joystickDeltaY) > 0.1)) {
                dx = joystickDeltaX;
                dy = joystickDeltaY;
            }

            if (dx !== 0 || dy !== 0) {
                const len = Math.hypot(dx, dy);
                dx /= len;
                dy /= len;

                gameState.playerX += dx * gameState.playerSpeed;
                gameState.playerY += dy * gameState.playerSpeed;
            }

            gameState.playerX = Math.max(50, Math.min(WORLD_WIDTH - 50, gameState.playerX));
            gameState.playerY = Math.max(50, Math.min(WORLD_HEIGHT - 50, gameState.playerY));

            let nearestBuilding = null;
            for (let b of BUILDINGS) {
                const bx = b.x + b.w / 2;
                const by = b.y + b.h / 2;
                const dist = Math.hypot(gameState.playerX - bx, gameState.playerY - by);
                if (dist < 180) {
                    nearestBuilding = b;
                    break;
                }
            }

            activeBuilding = nearestBuilding;
            const promptEl = document.getElementById('interaction-prompt');
            if (nearestBuilding) {
                promptEl.classList.remove('hidden');
                document.getElementById('prompt-location-name').innerText = nearestBuilding.name;
            } else {
                promptEl.classList.add('hidden');
            }
        }

        function updatePlayerInterior() {
            let dx = 0;
            let dy = 0;

            if (keys['w'] || keys['arrowup']) dy -= 1;
            if (keys['s'] || keys['arrowdown']) dy += 1;
            if (keys['a'] || keys['arrowleft']) dx -= 1;
            if (keys['d'] || keys['arrowright']) dx += 1;

            if (touchActive && (Math.abs(joystickDeltaX) > 0.1 || Math.abs(joystickDeltaY) > 0.1)) {
                dx = joystickDeltaX;
                dy = joystickDeltaY;
            }

            if (dx !== 0 || dy !== 0) {
                const len = Math.hypot(dx, dy);
                dx /= len;
                dy /= len;

                gameState.interiorPlayerX += dx * gameState.playerSpeed;
                gameState.interiorPlayerY += dy * gameState.playerSpeed;
            }

            gameState.interiorPlayerX = Math.max(100, Math.min(canvas.width - 100, gameState.interiorPlayerX));
            gameState.interiorPlayerY = Math.max(100, Math.min(canvas.height - 100, gameState.interiorPlayerY));

            const exitDist = Math.hypot(gameState.interiorPlayerX - (canvas.width / 2), gameState.interiorPlayerY - (canvas.height - 100));
            const promptEl = document.getElementById('interaction-prompt');
            promptEl.classList.remove('hidden');
            if (exitDist < 120) {
                document.getElementById('prompt-location-name').innerText = 'Exit to Open World';
            } else {
                document.getElementById('prompt-location-name').innerText = 'Interact with Counter / Room';
            }
        }

        function updateNPCs() {
            for (let npc of npcs) {
                npc.x += npc.vx;
                npc.y += npc.vy;
                if (npc.x < 100 || npc.x > WORLD_WIDTH - 100) npc.vx *= -1;
                if (npc.y < 100 || npc.y > WORLD_HEIGHT - 100) npc.vy *= -1;
            }
        }

        function updateVehicles() {
            for (let v of vehicles) {
                v.x += v.speed;
                if (v.x < -100) v.x = WORLD_WIDTH + 100;
                if (v.x > WORLD_WIDTH + 100) v.x = -100;
            }
        }

        function updateHUD() {
            const h = String(gameState.hour % 12 || 12).padStart(2, '0');
            const m = String(gameState.minute).padStart(2, '0');
            const ampm = gameState.hour >= 12 ? 'PM' : 'AM';
            document.getElementById('hud-clock').innerText = `${h}:${m} ${ampm}`;
            document.getElementById('hud-day').innerText = gameState.day;
            document.getElementById('hud-level').innerText = gameState.level;
            document.getElementById('hud-wallet').innerText = '₹' + gameState.wallet.toLocaleString();
            document.getElementById('hud-limit').innerText = `₹${Math.max(0, gameState.dailyLimit - gameState.dailySpent)} left`;
            document.getElementById('hud-energy-text').innerText = `${gameState.energy}%`;
            document.getElementById('hud-energy-bar').style.width = `${gameState.energy}%`;

            const curMission = MISSIONS[gameState.currentMissionId];
            if (curMission) {
                document.getElementById('hud-mission-title').innerText = curMission.title;
            }
        }

        function renderWorld() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            ctx.save();
            const camX = canvas.width / 2 - gameState.playerX;
            const camY = canvas.height / 2 - gameState.playerY;
            ctx.translate(camX, camY);

            ctx.fillStyle = '#0f172a';
            ctx.fillRect(0, 0, WORLD_WIDTH, WORLD_HEIGHT);

            ctx.fillStyle = '#1e293b';
            ctx.fillRect(0, 850, WORLD_WIDTH, 120);
            ctx.fillRect(0, 1750, WORLD_WIDTH, 120);
            ctx.fillRect(1350, 0, 120, WORLD_HEIGHT);
            ctx.fillRect(2350, 0, 120, WORLD_HEIGHT);

            for (let b of BUILDINGS) {
                ctx.fillStyle = 'rgba(0, 0, 0, 0.4)';
                ctx.fillRect(b.x + 10, b.y + 10, b.w, b.h);

                ctx.fillStyle = b.color;
                ctx.beginPath();
                ctx.roundRect(b.x, b.y, b.w, b.h, 16);
                ctx.fill();
                ctx.lineWidth = 3;
                ctx.strokeStyle = '#ffffff22';
                ctx.stroke();

                ctx.fillStyle = '#0f172a';
                ctx.fillRect(b.x + 20, b.y + 20, b.w - 40, 40);
                ctx.fillStyle = '#ffffff';
                ctx.font = 'bold 14px Outfit';
                ctx.textAlign = 'center';
                ctx.fillText(b.icon + ' ' + b.name, b.x + b.w / 2, b.y + 45);
            }

            for (let npc of npcs) {
                ctx.fillStyle = npc.color;
                ctx.beginPath();
                ctx.arc(npc.x, npc.y, 12, 0, Math.PI * 2);
                ctx.fill();
            }

            for (let v of vehicles) {
                ctx.fillStyle = v.color;
                ctx.beginPath();
                ctx.roundRect(v.x, v.y, 60, 30, 8);
                ctx.fill();
            }

            // Player Narendra Soni
            ctx.fillStyle = '#6366f1';
            ctx.shadowColor = '#6366f1';
            ctx.shadowBlur = 15;
            ctx.beginPath();
            ctx.arc(gameState.playerX, gameState.playerY, 16, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;

            ctx.fillStyle = '#ffffff';
            ctx.font = 'bold 12px Outfit';
            ctx.textAlign = 'center';
            ctx.fillText(gameState.role === 'warden' ? 'Warden Sharma' : 'Narendra Soni', gameState.playerX, gameState.playerY - 24);

            ctx.restore();

            if (gameState.hour >= 19 || gameState.hour < 6) {
                ctx.fillStyle = 'rgba(2, 6, 23, 0.45)';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
            }
        }

            ctx.font = 'bold 12px Outfit';
            ctx.textAlign = 'center';
            ctx.fillText('Narendra Soni', gameState.interiorPlayerX, gameState.interiorPlayerY - 26);
        }

        requestAnimationFrame(gameLoop);
